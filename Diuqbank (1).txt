const QUESTIONS = QRAW.map(r => ({
  id: r[0], dept: r[1], dept_full: r[2],
  course: r[3], sem: r[4], type: r[5], url: r[6]
}));

const DEPTS = [...new Set(QRAW.map(r => r[1]))].map(d => {
  const items = QRAW.filter(r => r[1] === d);
  return { short: d, full: items[0][2], count: items.length };
}).sort((a,b) => b.count - a.count);

const SEMESTERS = [...new Set(QRAW.map(r => r[4]))].sort((a,b) => {
  // Sort by year desc then season
  const parse = s => { const p = s.split(' '); return parseInt(p[1]||'0')*10 + (['Spring','Summer','Fall'].indexOf(p[0])); };
  return parse(b) - parse(a);
});

// ── STATE ───────────────────────────────────────────────
let state = {
  page: 'all',       // 'all' | 'dept:CSE' | 'settings'
  deptFilter: null,
  typeFilter: null,
  semFilter: '',
  search: '',
  currentPage: 1,
  pageSize: 24,
  currentQ: null,
  apiKey: localStorage.getItem('claude_api_key') || '',
};

// ── INIT ────────────────────────────────────────────────
function init() {
  buildSidebar();
  buildSemFilter();
  renderContent();
  document.getElementById('searchInput').addEventListener('input', e => {
    state.search = e.target.value;
    state.currentPage = 1;
    renderContent();
  });
}

// ── SIDEBAR ─────────────────────────────────────────────
function buildSidebar() {
  const nav = document.getElementById('sidebarNav');
  nav.innerHTML = `
    <div class="nav-section-label">Browse</div>
    <div class="nav-item ${state.page==='all'?'active':''}" onclick="navigate('all')">
      <span>📚</span> All Questions
      <span class="count">${QUESTIONS.length}</span>
    </div>
    <div class="nav-item ${state.page==='settings'?'active':''}" onclick="navigate('settings')">
      <span>⚙️</span> Settings
    </div>
    <div class="nav-section-label" style="margin-top:8px">Departments</div>
    ${DEPTS.map(d => `
      <div class="nav-item ${state.page==='dept:'+d.short?'active':''}" onclick="navigate('dept:${d.short}')">
        <span class="dept-dot dot-${d.short}"></span>
        <span>${d.short}</span>
        <span class="count">${d.count}</span>
      </div>
    `).join('')}
  `;
}

function buildSemFilter() {
  const sel = document.getElementById('semFilter');
  SEMESTERS.forEach(s => {
    const opt = document.createElement('option');
    opt.value = s; opt.textContent = s;
    sel.appendChild(opt);
  });
}

// ── NAVIGATION ──────────────────────────────────────────
function navigate(page) {
  state.page = page;
  state.currentPage = 1;
  if (page.startsWith('dept:')) {
    state.deptFilter = page.split(':')[1];
  } else if (page === 'all') {
    state.deptFilter = null;
  }
  buildSidebar();
  renderContent();
}

function setTypeFilter(type, el) {
  state.typeFilter = type;
  state.currentPage = 1;
  document.querySelectorAll('.filter-chip').forEach(c => c.classList.remove('active'));
  el.classList.add('active');
  renderContent();
}

function setSemFilter(val) {
  state.semFilter = val;
  state.currentPage = 1;
  renderContent();
}

// ── FILTER ───────────────────────────────────────────────
function getFiltered() {
  let qs = QUESTIONS;
  if (state.deptFilter) qs = qs.filter(q => q.dept === state.deptFilter);
  if (state.typeFilter) qs = qs.filter(q => q.type === state.typeFilter);
  if (state.semFilter)  qs = qs.filter(q => q.sem === state.semFilter);
  if (state.search) {
    const s = state.search.toLowerCase();
    qs = qs.filter(q => q.course.toLowerCase().includes(s) || q.sem.toLowerCase().includes(s) || q.dept.toLowerCase().includes(s) || q.dept_full.toLowerCase().includes(s));
  }
  return qs;
}

// ── RENDER ───────────────────────────────────────────────
function renderContent() {
  const content = document.getElementById('content');

  if (state.page === 'settings') {
    document.getElementById('topbarTitle').textContent = 'Settings';
    document.getElementById('topbarSub').textContent = 'Configure your API key and preferences';
    document.getElementById('filterbar').style.display = 'none';
    renderSettings(content);
    return;
  }

  document.getElementById('filterbar').style.display = '';

  const filtered = getFiltered();
  const total = filtered.length;
  const pages = Math.ceil(total / state.pageSize);
  const start = (state.currentPage - 1) * state.pageSize;
  const paged = filtered.slice(start, start + state.pageSize);

  // Update topbar
  const deptInfo = state.deptFilter ? DEPTS.find(d => d.short === state.deptFilter) : null;
  document.getElementById('topbarTitle').textContent = deptInfo ? `${deptInfo.short} — ${deptInfo.full}` : 'All Questions';
  document.getElementById('topbarSub').textContent = `${total} question papers found`;

  if (total === 0) {
    content.innerHTML = `
      <div class="empty-state">
        <div class="empty-icon">🔍</div>
        <div class="empty-title">No results found</div>
        <div class="empty-desc">Try adjusting your filters or search term.</div>
      </div>`;
    return;
  }

  // Stats
  const statsByType = {};
  filtered.forEach(q => { statsByType[q.type] = (statsByType[q.type]||0)+1; });

  let html = '';

  // Only show stats on 'all' or dept pages, not when heavily filtered
  if (!state.search && !state.typeFilter) {
    const depts_count = [...new Set(filtered.map(q=>q.dept))].length;
    const courses_count = [...new Set(filtered.map(q=>q.course))].length;
    html += `
    <div class="stats-row">
      <div class="stat-card">
        <div class="stat-label">Total Papers</div>
        <div class="stat-value" style="color:var(--accent)">${total}</div>
        <div class="stat-sub">Question papers</div>

      <div class="stat-card">
        <div class="stat-label">Courses</div>
        <div class="stat-value" style="color:var(--green)">${courses_count}</div>
        <div class="stat-sub">Unique courses</div>
      </div>
      <div class="stat-card">
        <div class="stat-label">Exam Types</div>
        <div class="stat-value" style="color:var(--orange)">${Object.keys(statsByType).length}</div>
        <div class="stat-sub">${Object.entries(statsByType).map(([k,v])=>k+': '+v).join(', ')}</div>
      </div>
    </div>`;
  }

  // Cards
  html += `<div class="cards-grid">`;
  paged.forEach(q => {
    const badgeClass = {Mid:'badge-mid',Final:'badge-final',Quiz:'badge-quiz','Lab Final':'badge-lab'}[q.type]||'badge-mid';
    html += `
    <div class="q-card" onclick="openQuestion(${q.id})">
      <div class="card-header">
        <div class="card-course">${escHtml(q.course)}</div>
        <span class="badge ${badgeClass}">${q.type}</span>
      </div>
      <div class="card-meta">
        <span class="meta-tag">📅 ${q.sem}</span>
        <span class="meta-tag">🏛️ ${q.dept}</span>
      </div>
      <div class="card-footer">
        <span class="badge card-dept-badge dept-${q.dept}">${q.dept}</span>
        <button class="btn-view" onclick="event.stopPropagation();openQuestion(${q.id})">View PDF →</button>
      </div>
    </div>`;
  });
  html += '</div>';

  // Pagination
  if (pages > 1) {
    html += `<div class="pagination">`;
    html += `<button class="page-btn" onclick="goPage(${state.currentPage-1})" ${state.currentPage<=1?'disabled':''}>← Prev</button>`;
    const range = getPaginationRange(state.currentPage, pages);
    range.forEach(p => {
      if (p === '…') html += `<span style="color:var(--muted); padding:0 4px;">…</span>`;
      else html += `<button class="page-btn ${p===state.currentPage?'active':''}" onclick="goPage(${p})">${p}</button>`;
    });
    html += `<button class="page-btn" onclick="goPage(${state.currentPage+1})" ${state.currentPage>=pages?'disabled':''}>Next →</button>`;
    html += `<span style="font-size:12px; color:var(--muted); margin-left:8px;">Page ${state.currentPage} of ${pages}</span>`;
    html += `</div>`;
  }

  content.innerHTML = html;
}

function getPaginationRange(cur, total) {
  if (total <= 7) return Array.from({length:total},(_,i)=>i+1);
  if (cur <= 4) return [1,2,3,4,5,'…',total];
  if (cur >= total-3) return [1,'…',total-4,total-3,total-2,total-1,total];
  return [1,'…',cur-1,cur,cur+1,'…',total];
}

function goPage(p) {
  const filtered = getFiltered();
  const pages = Math.ceil(filtered.length / state.pageSize);
  if (p < 1 || p > pages) return;
  state.currentPage = p;
  renderContent();
  document.getElementById('content').scrollTop = 0;
}

// ── MULTI-PROVIDER AI SYSTEM ──────────────────────────────

function detectProvider(key) {
  if (!key) return { name: 'Unknown', model: '', id: 'none' };
  if (key.startsWith('sk-ant-')) return { name: 'Anthropic Claude', model: 'claude-sonnet-4-20250514', id: 'anthropic' };
  if (key.startsWith('AIza'))    return { name: 'Google Gemini',    model: 'gemini-2.0-flash',          id: 'gemini' };
  if (key.startsWith('gsk_'))    return { name: 'Groq (Free)',       model: 'llama-3.3-70b-versatile',   id: 'groq' };
  if (key.startsWith('sk-'))     return { name: 'OpenAI GPT-4o',    model: 'gpt-4o',                    id: 'openai' };
  return { name: 'Unknown', model: '', id: 'none' };
}

async function callAI(key, prompt) {
  const p = detectProvider(key);
  if (p.id === 'none') throw new Error('Unrecognized API key format. Please check Settings.');

  if (p.id === 'anthropic') {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-api-key': key, 'anthropic-version': '2023-06-01', 'anthropic-dangerous-direct-browser-access': 'true' },
      body: JSON.stringify({ model: p.model, max_tokens: 1500, messages: [{ role: 'user', content: prompt }] })
    });
    const data = await res.json();
    if (data.error) throw new Error(data.error.message || JSON.stringify(data.error));
    return data.content[0].text;
  }

  if (p.id === 'openai') {
    const res = await fetch('https://api.openai.com/v1/chat/completions', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + key },
      body: JSON.stringify({ model: p.model, max_tokens: 1500, messages: [{ role: 'user', content: prompt }] })
    });
    const data = await res.json();
    if (data.error) throw new Error(data.error.message || JSON.stringify(data.error));
    return data.choices[0].message.content;
  }

  if (p.id === 'gemini') {
    const res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/${p.model}:generateContent?key=${key}`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contents: [{ parts: [{ text: prompt }] }] })
    });
    const data = await res.json();
    if (data.error) throw new Error(data.error.message || JSON.stringify(data.error));
    return data.candidates[0].content.parts[0].text;
  }

  if (p.id === 'groq') {
    const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + key },
      body: JSON.stringify({ model: p.model, max_tokens: 1500, messages: [{ role: 'user', content: prompt }] })
    });
    const data = await res.json();
    if (data.error) throw new Error(data.error.message || JSON.stringify(data.error));
    return data.choices[0].message.content;
  }
}

async function callAIPanel(prompt) {
  const btn = document.getElementById('solveBtn');
  const resultDiv = document.getElementById('aiResult');
  const p = detectProvider(state.apiKey);

  // Update label
  document.getElementById('aiProviderLabel').textContent = `Using ${p.name} (${p.model})`;

  btn.disabled = true;
  btn.textContent = '⏳ Thinking…';
  resultDiv.style.display = 'block';
  resultDiv.className = 'ai-response';
  resultDiv.innerHTML = '<div class="spinner" style="margin:0 auto"></div><p style="text-align:center;margin-top:10px;color:var(--muted)">AI is thinking…</p>';

  try {
    const text = await callAI(state.apiKey, prompt);
    resultDiv.className = 'ai-response';
    resultDiv.innerHTML = formatMarkdown(text);
  } catch(e) {
    resultDiv.className = 'ai-response ai-error';
    let msg = e.message || String(e);
    // Friendly guidance for common errors
    if (msg.includes('quota') || msg.includes('billing') || msg.includes('limit')) {
      msg += '\n\n💡 Tip: Your API quota is exhausted. Try switching to Groq (free) — get a key at console.groq.com';
    } else if (msg.includes('auth') || msg.includes('key') || msg.includes('invalid')) {
      msg += '\n\n💡 Tip: Check your API key in Settings. Make sure you pasted the full key.';
    }
    resultDiv.textContent = '❌ ' + msg;
  }

  btn.disabled = false;
  btn.textContent = '✨ Solve with AI';
}

// Simple markdown formatter
function formatMarkdown(text) {
  return text
    .replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;')
    .replace(/```(\w*)\n([\s\S]*?)```/g, '<pre style="background:rgba(255,255,255,.05);border:1px solid var(--border);border-radius:6px;padding:10px;overflow-x:auto;font-size:12px;margin:6px 0"><code>$2</code></pre>')
    .replace(/`([^`]+)`/g, '<code>$1</code>')
    .replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>')
    .replace(/\*([^*]+)\*/g, '<em>$1</em>')
    .replace(/^### (.+)$/gm, '<h3>$1</h3>')
    .replace(/^## (.+)$/gm, '<h3>$1</h3>')
    .replace(/^# (.+)$/gm, '<h3>$1</h3>')
    .replace(/^- (.+)$/gm, '<li style="margin:3px 0;padding-left:4px">• $1</li>')
    .replace(/^\d+\. (.+)$/gm, '<li style="margin:3px 0;padding-left:4px">$1</li>')
    .replace(/\n\n/g, '<br/><br/>')
    .replace(/\n/g, '<br/>');
}

function escHtml(s) {
  return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
